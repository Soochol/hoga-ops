def measure(fn,n=5):
    obj=fn();times=[]
    for _ in range(n):
        t=time.perf_counter();v=fn();times.append((time.perf_counter()-t)*1000)
        assert v==obj
    return round(statistics.median(times),3),obj

def size(obj):
    if isinstance(obj,list):obj=[x.model_dump() if hasattr(x,'model_dump') else x for x in obj]
    elif hasattr(obj,'model_dump'):obj=obj.model_dump()
    return len(json.dumps(obj,separators=(',',':')).encode())
