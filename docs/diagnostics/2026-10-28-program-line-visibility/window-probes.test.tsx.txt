

describe('diagnostic: program window chart-source switching', () => {
  it('minute -> daily -> seconds -> minute removes and restores lines without changing the stock', async () => {
    const { useProgramTradeDisplayStore } = await import('../../state/programTradeDisplay');
    useProgramTradeDisplayStore.setState({ view: 'intraday', measure: 'amount' });
    __resetGroupChartLinksForTests();
    useLiveCursorStore.getState().resetCursor();
    const t = Date.parse('2026-07-20T09:00:00+09:00');
    const bundle = {
      code: '005930',
      candles: [0, 60_000].map((offset, i) => ({ ts_ms: t + offset, open: 70000, high: 70500, low: 70000, close: 70000 + i * 500, vol_a: 10, vol_b: 0 })),
      program_trade: { source: 'kis_program_trade', points: [0, 30_000].map((offset, i) => ({ t: t + offset, net_qty: 10 + i, net_amount: 100_000_000 * (i + 1), gap_risk: false })) },
    } as unknown as RangeBundle;
    publishGroupChartLink(chartLink({ bundle }));
    const view = renderWithQuery(<DataWindow win={dataWin('program')} symbol={{ code: '005930', name: '삼성전자' }} />);
    expect(screen.getByTestId('program-price-graph')).toBeInTheDocument();
    act(() => publishGroupChartLink(chartLink({ windowId: 'daily', timeframe: 'D', bundle: { ...bundle, candles: bundle.candles.slice(0, 1) } })));
    expect(screen.getByTestId('program-sparkline')).toBeInTheDocument();
    expect(screen.queryByTestId('program-price-graph')).toBeNull();
    act(() => publishGroupChartLink(chartLink({ windowId: 'seconds', timeframe: '1s', bundle: null })));
    expect(screen.queryByTestId('program-sparkline')).toBeNull();
    expect(screen.getByText('프로그램 순매수 데이터 없음')).toBeInTheDocument();
    act(() => publishGroupChartLink(chartLink({ bundle })));
    expect(screen.getByTestId('program-price-graph')).toBeInTheDocument();
    expect(screen.getByTestId('program-sparkline')).toBeInTheDocument();
    view.unmount();
    __resetGroupChartLinksForTests();
  });
});


it('diagnostic: after reload or stock switch, ready candles remain hidden until program history arrives', async () => {
  const { useProgramTradeDisplayStore } = await import('../../state/programTradeDisplay');
  useProgramTradeDisplayStore.setState({ view: 'intraday', measure: 'amount' });
  __resetGroupChartLinksForTests();
  useLiveCursorStore.getState().resetCursor();
  const t = Date.parse('2026-07-20T09:00:00+09:00');
  const candles = [0, 60_000].map((offset, i) => ({ ts_ms: t + offset, open: 70000, high: 70500, low: 70000, close: 70000 + i * 500, vol_a: 10, vol_b: 0 }));
  const program = { source: 'kis_program_trade', points: [0, 30_000].map((offset, i) => ({ t: t + offset, net_qty: 10 + i, net_amount: 100_000_000 * (i + 1), gap_risk: false })) };
  const full = { code: '005930', candles, program_trade: program } as unknown as RangeBundle;
  publishGroupChartLink(chartLink({ bundle: full }));
  const view = renderWithQuery(<DataWindow win={dataWin('program')} symbol={{ code: '005930', name: '삼성전자' }} />);
  expect(screen.getByTestId('program-price-graph')).toBeInTheDocument();
  // Stock switch: the chart link still belongs to the previous stock for a commit.
  view.rerender(<QueryClientProvider client={new QueryClient({ defaultOptions: { queries: { retry: false } } })}><DataWindow win={dataWin('program')} symbol={{ code: '000660', name: 'SK하이닉스' }} /></QueryClientProvider>);
  expect(screen.getByText(/차트 창 연동 대기/)).toBeInTheDocument();
  // New code/reload with neither response ready.
  act(() => publishGroupChartLink(chartLink({ code: '000660', bundle: { ...full, code: '000660', candles: [], program_trade: { points: [] } } })));
  expect(screen.getByText('프로그램 순매수 데이터 없음')).toBeInTheDocument();
  // Candles complete first: both close samples exist but the price line is hidden.
  act(() => publishGroupChartLink(chartLink({ code: '000660', bundle: { ...full, code: '000660', program_trade: { points: [] } } })));
  expect(screen.queryByTestId('program-price-graph')).toBeNull();
  expect(screen.getByText('프로그램 순매수 데이터 없음')).toBeInTheDocument();
  // Program history completes: both appear together without another interaction.
  act(() => publishGroupChartLink(chartLink({ code: '000660', bundle: { ...full, code: '000660' } })));
  expect(screen.getByTestId('program-price-graph')).toBeInTheDocument();
  expect(screen.getByTestId('program-sparkline')).toBeInTheDocument();
  // Reverse order on a fresh load: program first displays alone, then price joins.
  act(() => publishGroupChartLink(chartLink({ code: '000660', bundle: { ...full, code: '000660', candles: [] } })));
  expect(screen.getByTestId('program-sparkline')).toBeInTheDocument();
  expect(screen.queryByTestId('program-price-graph')).toBeNull();
  act(() => publishGroupChartLink(chartLink({ code: '000660', bundle: { ...full, code: '000660' } })));
  expect(screen.getByTestId('program-price-graph')).toBeInTheDocument();
  view.unmount();
  __resetGroupChartLinksForTests();
});
