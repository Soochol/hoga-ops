

describe('diagnostic: intermittent program line visibility', () => {
  const start = T0 + 30 * 60_000;
  const series = seriesOf([point(start, 100_000_000), point(start + 30_000, 200_000_000)]);
  const closes = [{ t_ms: T0, close: 70000 }, { t_ms: T0 + 60 * 60_000, close: 70500 }];

  it('two valid close points disappear when program points temporarily fall below two, then return', () => {
    const view = render(<ProgramTradeSummaryCard series={series} closePoints={closes} todayKst="20260721" />);
    expect(screen.getByTestId('program-price-graph')).toBeInTheDocument();
    view.rerender(<ProgramTradeSummaryCard series={seriesOf([series.points[0]])} closePoints={closes} todayKst="20260721" />);
    expect(screen.queryByTestId('program-price-graph')).toBeNull();
    expect(screen.queryByTestId('program-sparkline')).toBeNull();
    expect(screen.getByTestId('program-metric-strip')).toBeInTheDocument();
    view.rerender(<ProgramTradeSummaryCard series={series} closePoints={closes} todayKst="20260721" />);
    expect(screen.getByTestId('program-price-graph')).toBeInTheDocument();
  });

  it('daily candle input removes the price line even though program history stays complete', () => {
    const view = render(<ProgramTradeSummaryCard series={series} closePoints={closes} todayKst="20260721" />);
    expect(screen.getByTestId('program-price-graph')).toBeInTheDocument();
    view.rerender(<ProgramTradeSummaryCard series={series} closePoints={[closes[0]]} todayKst="20260721" />);
    expect(screen.queryByTestId('program-price-graph')).toBeNull();
    expect(screen.getByTestId('program-sparkline')).toBeInTheDocument();
    view.rerender(<ProgramTradeSummaryCard series={series} closePoints={closes} todayKst="20260721" />);
    expect(screen.getByTestId('program-price-graph')).toBeInTheDocument();
  });

  it('a cursor on a day without program observations hides both lines and resetting it restores them', () => {
    const view = render(<ProgramTradeSummaryCard series={series} closePoints={closes} todayKst="20260721" />);
    expect(screen.getByTestId('program-price-graph')).toBeInTheDocument();
    view.rerender(<ProgramTradeSummaryCard series={series} closePoints={closes} todayKst="20260721" cursorMs={T0 - DAY_MS} />);
    expect(screen.queryByTestId('program-sparkline')).toBeNull();
    view.rerender(<ProgramTradeSummaryCard series={series} closePoints={closes} todayKst="20260721" />);
    expect(screen.getByTestId('program-price-graph')).toBeInTheDocument();
  });

  it('past-day local hover before the first observation unmounts the graph and leaves a sticky cursor', () => {
    const view = render(<ProgramTradeSummaryCard series={series} closePoints={closes} todayKst="20260722" cursorMs={start} />);
    const graph = screen.getByTestId('program-sparkline');
    vi.spyOn(graph, 'getBoundingClientRect').mockReturnValue(new DOMRect(0, 0, 1000, 100));
    fireEvent.mouseMove(graph, { clientX: 100, clientY: 50 });
    expect(screen.queryByTestId('program-sparkline')).toBeNull();
    expect(screen.getByText('프로그램 순매수 데이터 없음')).toBeInTheDocument();
    view.rerender(<ProgramTradeSummaryCard series={series} closePoints={closes} todayKst="20260722" cursorMs={start + 30_000} />);
    expect(screen.queryByTestId('program-sparkline')).toBeNull();
  });
});
