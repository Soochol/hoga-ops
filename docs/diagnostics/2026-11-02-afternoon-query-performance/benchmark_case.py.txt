def norm(x):
 if isinstance(x,BaseModel):return ['model',type(x).__name__,norm(x.model_dump())]
 if dataclasses.is_dataclass(x):return ['dataclass',type(x).__name__,[[f.name,norm(getattr(x,f.name))] for f in dataclasses.fields(x)]]
 if isinstance(x,dict):return ['dict',sorted([[norm(k),norm(v)] for k,v in x.items()],key=lambda p:repr(p[0]))]
 if isinstance(x,(set,frozenset)):return ['set',sorted([norm(v) for v in x],key=repr)]
 if isinstance(x,(list,tuple)):return [type(x).__name__,[norm(v) for v in x]]
 if isinstance(x,(dt.date,dt.datetime)):return [type(x).__name__,x.isoformat()]
 return [type(x).__name__,x]
