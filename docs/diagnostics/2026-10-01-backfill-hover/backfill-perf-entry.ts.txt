// Diagnostic-only timing of canonical query cache publication.
import {QueryClient} from '@tanstack/react-query';
const originalSet = QueryClient.prototype.setQueryData;
QueryClient.prototype.setQueryData = function(key, updater, options) {
 const p=window.__bf; const t=performance.now();
 if(window.__bfNoSharing && key[0]==='live' && key[1]==='past-candles' && key[2]==='merged')this.setQueryDefaults(key,{structuralSharing:false});
 try{return originalSet.call(this,key,updater,options);}
 finally{if(p)p.cacheWrites.push({key,t,dt:performance.now()-t});}
};
